% Numbers 1-1000 to cross out while counting voters: 500 per A4 page, in
% rows of 20, with a heavier line after every 100 and between the tens.
\documentclass[10pt]{article}
\usepackage[a4paper,margin=15mm,footskip=8mm]{geometry}
\usepackage{fontspec}
\usepackage{xcolor}
\usepackage{tikz}
\usepackage{fancyhdr}

\setmainfont{Lato}
\newfontfamily\blackfont{Lato Black}

\definecolor{ink}{HTML}{1F2937}
\definecolor{accent}{HTML}{1E3A5F}
\definecolor{muted}{HTML}{6B7280}
\definecolor{hairline}{HTML}{D1D5DB}

\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0pt}
\fancyfoot[R]{\footnotesize\color{muted}\thepage\,/\,2}

\newcommand{\cellw}{8.97mm}
\newcommand{\cellh}{9.6mm}

% \numbers{FIRST}: a 20 x 25 grid starting at FIRST.
\newcommand{\numbers}[1]{%
  \begin{tikzpicture}[x=\cellw, y=-\cellh]
    \foreach \r in {0,...,24} {
      \foreach \c in {0,...,19} {
        \pgfmathtruncatemacro{\n}{#1 + 20*\r + \c}
        \node[font=\small, text=ink, text height=1.6ex, text depth=0pt] at (\c+0.5, \r+0.5) {\n};
      }
    }
    \foreach \r in {1,...,24} \draw[hairline, line width=0.4pt] (0,\r) -- (20,\r);
    \foreach \c in {1,...,19} \draw[hairline, line width=0.4pt] (\c,0) -- (\c,25);
    \foreach \r in {0,5,...,25} \draw[accent, line width=1.2pt] (0,\r) -- (20,\r);
    \foreach \c in {0,10,20} \draw[accent, line width=1.2pt] (\c,0) -- (\c,25);
  \end{tikzpicture}}

% \sheet{FIRST}{LAST}: one page with title and grid.
\newcommand{\sheet}[2]{%
  \noindent{\blackfont\fontsize{20}{24}\selectfont Brojanje birača}%
  \hfill{\large\color{muted}#1\,--\,#2}\par
  \vspace{2pt}\noindent{\color{accent}\rule{2.5em}{2.5pt}}\par
  \vspace{1.2em}\noindent\numbers{#1}\newpage}

\begin{document}
\color{ink}
\sheet{1}{500}
\sheet{501}{1000}
\end{document}
